


                # login frame
        frame = tk.Frame()
        frame.columnconfigure(0, weight=0)
        frame.columnconfigure(0, weight=2)

        # row 1 username
        tk.Label(frame, text='Username: ').grid(column=0, row=0, sticky=tk.W)
        username = tk.Entry(frame, width=30)
        username.focus()
        username.grid(column=1, row=0, sticky=tk.W)

        #row 2 password
        tk.Label(frame, text='Password: ').grid(column=0, row=1, sticky=tk.W)
        password = tk.Entry(frame, width=30)
        password.grid(column=1, row=1, sticky=tk.W)
